import java.lang.reflect.Field;
import java.lang.reflect.Modifier;
import java.util.Arrays;
import java.util.TreeMap;

/** Probe do gia tri HIEU DUNG cua Configs/LiveProfileC3 (reflection, chi doc). KHONG phai ma san pham. */
public class Probe {
    public static void main(String[] a) throws Exception {
        TreeMap<String, String> out = new TreeMap<>();
        String[] classes = {"com.binance.chuyennd.tradecore.Configs",
                "com.binance.chuyennd.tradecore.selector.LiveProfileC3"};
        for (String cn : classes) {
            Class<?> c = Class.forName(cn);
            String sn = c.getSimpleName();
            for (Field f : c.getDeclaredFields()) {
                if (!Modifier.isStatic(f.getModifiers())) continue;
                Class<?> t = f.getType();
                if (!(t.isPrimitive() || t == Float.class || t == Integer.class || t == Long.class
                        || t == Boolean.class || t == Double.class || t == String.class || t == float[].class)) continue;
                f.setAccessible(true);
                Object v;
                try { v = f.get(null); } catch (Throwable e) { v = "ERR " + e; }
                String s = (v instanceof float[]) ? Arrays.toString((float[]) v) : String.valueOf(v);
                if (s.length() > 120) s = s.substring(0, 120) + "...";
                out.put(sn + "." + f.getName(), s);
            }
        }
        for (java.util.Map.Entry<String, String> e : out.entrySet()) System.out.println("PROBE\t" + e.getKey() + "\t" + e.getValue());
        System.out.println("PROBE_DCA_TOTAL_WEIGHT\t" + com.binance.chuyennd.tradecore.Configs.dcaGridTotalWeight());
        System.out.println("PROBE_TS_PNOPUMP_WEAK_THR\t" + com.binance.chuyennd.tradecore.Configs.tsPnoPumpWeakThr());
        System.exit(0);
    }
}
